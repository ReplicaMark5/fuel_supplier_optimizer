Data Values:

[ij] with i being depots and j being suppliers

V[i] = Depot Predicted Volume 
COC[i,j] = Rebate Rand per litre for collection
DEL[i,j] = Rebate Rand per Litre for delivery
Cost[i,j] = Cost of Collection in Rand per litre
DP = Diesel Price 
ZD = Zone Differential 

S[j] = Supplier Score


Decision Variables:
C[i,j] = COC option 
D[i,j] = DEL option 



max z = (S[Supplier 1])((C11 +C21 +C31 +C41 +C51) + (D11 +D21 +D31 +D41 +D51))
      + (S[supplier 2])((C12 +C22 +C32 +C42 +C52) + (D12 +D22 +D32 +D42 +D52))

# Maximize supplier score objective function
max_score = mdl.sum(
    S[f"Supplier {j}"] * (C[i, j] + D[i, j])
    for i in range(1, 6)  # depots
    for j in range(1, 3)  # suppliers
)
mdl.maximize(max_score)
